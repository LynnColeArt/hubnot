func TestProbeMissingProgressDoesNotRetryUncertainCreate(t *testing.T) {
 ctx:=context.Background()
 snap, blobs:=migrationForeignSnapshot()
 d:=&migrationReceiptDestination{locator:migration.Locator{Kind:"fixture",Repository:"d"}}
 s,e:=PrepareMigration(ctx,filepath.Join(t.TempDir(),"session"),snap,blobs,migration.Selection{Destination:d.locator},d,migration.DefaultLimits(),nil)
 if e!=nil { t.Fatal(e) }
 if e=ApplyMigration(ctx,s,d,"actor");e==nil {t.Fatal("expected uncertain comment")}
 if d.creates!=1 || d.comments!=1 {t.Fatal("setup writes",d.creates,d.comments)}
 if e=os.Remove(filepath.Join(s.Directory,"progress.json"));e!=nil {t.Fatal(e)}
 e=ApplyMigration(ctx,s,d,"actor")
 if d.creates!=1 || d.comments!=1 {t.Fatalf("missing authenticated checkpoint repeated uncertain creates: issues=%d comments=%d err=%v",d.creates,d.comments,e)}
}
